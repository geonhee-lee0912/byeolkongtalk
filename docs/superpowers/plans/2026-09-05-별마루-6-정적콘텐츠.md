  import cardLines from "@/data/byeolmaru/card-lines.json";
  export function getCardLine(cardId: number, reversed: boolean): string | null {
    const e = (cardLines as Record<string, { upright: string; reversed: string }>)[String(cardId)];
    if (!e) return null;               // 뱅크 미스 → 호출측 폴백

  }

### Task 3: 골격 45 뱅크 저작 + 조회(로테이션) + 계약

**Files:** Create `data/byeolmaru/skeleton-lines.json` · Modify `lib/byeolmaru/static-lines.ts` · `lib/byeolmaru/static-lines.test.ts`

- [ ] **Step 1: skeleton-lines.json 저작(세션 내)** — tone(3) × relation(5) × variant 3 = 45. 각 문장 = tone valence + relation flavor(생아=날 살려줌 / 극아=날 누름 / 비화=같은 결 / 아극=내가 다룸 / 아생=내가 내줌), 단정 금지.
  ```json
  { "good": { "생아": ["v0","v1","v2"], "아극": [...], "비화": [...], "아생": [...], "극아": [...] },
    "normal": { ... }, "caution": { ... } }
  ```
- [ ] **Step 2: getSkeletonLine 구현(날짜 해시 로테이션)** — 결정론적(같은 date→같은 variant):
  ```ts
  import skeletonLines from "@/data/byeolmaru/skeleton-lines.json";
  import type { ElementRelation } from "@/lib/saju/pairing";
  import type { DayTone } from "./day-score";
  function hashDate(date: string): number { let h = 0; for (const c of date) h = (h * 31 + c.charCodeAt(0)) | 0; return Math.abs(h); }
  export function getSkeletonLine(tone: DayTone, relation: ElementRelation, date: string): string | null {
    const arr = (skeletonLines as Record<string, Record<string, string[]>>)[tone]?.[relation];
    if (!arr || arr.length === 0) return null;
    return arr[hashDate(date) % arr.length];
  }
  ```
- [ ] **Step 3: 계약 테스트** — 3 tone × 5 relation 전부 존재 · 각 variant ≥3 · 비어있지 않음 · 금지문자 없음 · **로테이션 결정론**(같은 date 두 번 = 같은 결과) · 서로 다른 date 가 variant 다양성 커버.

```ts
const TONES = ["good","normal","caution"] as const;
const RELS = ["생아","아극","비화","아생","극아"] as const;
test("skeleton: 15조합 전부 · variant≥3 · 결정론", () => {
  for (const t of TONES) for (const r of RELS) {
    const a = getSkeletonLine(t, r, "2026-09-05");
    assert.ok(a && a.length > 0, `${t}/${r}`);
    assert.equal(getSkeletonLine(t, r, "2026-09-05"), a, "결정론");
    assert.ok(!a.includes("**"));
  }
});
```

- [ ] **Step 4: 통과 확인** — Run test → PASS.
- [ ] **Step 5: Commit** — `feat(byeolmaru): 골격 45 뱅크 + getSkeletonLine(날짜 로테이션) (⑥ T3)`
