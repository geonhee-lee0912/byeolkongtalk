import { canCalcSaju, type SajuInput } from "./calc.ts";

// ── 계약: 표 = 엔진 ──
// 탐침은 calcSaju(canCalcSaju)만 쓴다 — tyme4ts 직접 호출 금지(AGENTS.md). 결과는 표와 같은 [윤달, 대소월] 형식.
const lunar = (year: number, month: number, day: number, isLeapMonth: boolean): SajuInput => ({
  year,
  month,
  day,
  hour: null,
  isLunar: true,
  isLeapMonth,
  gender: "other",
});

function probeYear(year: number): [number, string] {
  const leaps: number[] = [];
  for (let m = 1; m <= 12; m++) if (canCalcSaju(lunar(year, m, 1, true))) leaps.push(m);
  assert.ok(leaps.length <= 1, `${year}: 윤달이 둘 이상 — ${leaps.join(",")}`);
  const leap = leaps[0] ?? 0;
  let bits = "";
  for (let m = 1; m <= 12; m++) {
    for (const isLeap of leap === m ? [false, true] : [false]) {
      const tag = `${year}-${m}${isLeap ? "(윤)" : ""}`;
      // 음력 달은 29일 아니면 30일 — 29일은 늘 있고 31일은 늘 없다.
      assert.ok(canCalcSaju(lunar(year, m, 29, isLeap)), `${tag}: 29일이 없다`);
      assert.ok(!canCalcSaju(lunar(year, m, 31, isLeap)), `${tag}: 31일이 있다`);
      bits += canCalcSaju(lunar(year, m, 30, isLeap)) ? "1" : "0";
    }
  }
  return [leap, bits];

//    리터럴 전체를 실패 메시지에 찍는다(tyme4ts 업그레이드 시 그대로 붙여넣을 것).

export const LUNAR_TABLE_START = 1900;
export const LUNAR_TABLE_END = 2100;

// [그해 윤달 월(0=없음), 대소월] — 대소월은 그해 달 순서대로(윤달은 그 월 바로 뒤) '1'=30일 · '0'=29일.
// 인덱스 = 연도 − LUNAR_TABLE_START.
const TABLE: readonly (readonly [number, string])[] = [
  // @@LUNAR_TABLE@@ — 계약 테스트(lunar-table.test.ts)가 찍는 리터럴로 채운다
];

/** 그해 윤달 월(1~12). 없으면 0. 표 범위 밖도 0. */
export function lunarLeapMonth(year: number): number {
  return TABLE[year - LUNAR_TABLE_START]?.[0] ?? 0;
}

/** 음력 그달 일수(29|30). 없는 달(그해 윤달이 아닌데 isLeap, 표 범위 밖)은 0. */
export function lunarMonthDays(year: number, month: number, isLeap: boolean): number {
  const row = TABLE[year - LUNAR_TABLE_START];
  if (!row || month < 1 || month > 12) return 0;
  const [leap, bits] = row;
  if (isLeap && leap !== month) return 0;
  // 윤달이 있는 해는 윤달과 그 뒤 달들이 문자열에서 한 칸씩 밀린다.
  const shifted = leap !== 0 && (month > leap || (isLeap && month === leap));
  return bits[month - 1 + (shifted ? 1 : 0)] === "1" ? 30 : 29;
}

/**
 * 생년월일 폼이 보여줄 마지막 날.
 * - 양력: 그 월 일수.
 * - 음력: min(음력 그달 일수, 같은 숫자 월의 양력 일수). 🔴 뒤쪽은 user_profiles.birth_date 가 DATE 컬럼이라서다 —
 *   음력도 그레고리력에 있는 Y-M-D 만 저장된다(음력 2/30 → 22008). isLeap 은 그해 윤달이 이 달일 때만 적용.
 */
export function maxBirthDay(year: number, month: number, isLunar: boolean, isLeap: boolean): number {
  const solarDays = new Date(year, month, 0).getDate(); // month 는 1-based → 다음 달 0일 = 이 달 말일
  if (!isLunar) return solarDays;
  const leap = isLeap && lunarLeapMonth(year) === month;
  return Math.min(lunarMonthDays(year, month, leap), solarDays);
}

## 부록 A — 브라우저 Dropdown 헬퍼 (`javascript_tool` 로 설치, 페이지 이동마다 다시)

`components/common/Dropdown` 은 트리거 `button[aria-haspopup="listbox"][aria-label=…]` + 목록 `[role="listbox"][aria-label=…] [role="option"]` 구조다(옵션 문구 `1996년`·`4월`·`29일`).

```js
(async () => {
  const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
  const trigger = (label) => document.querySelector(`button[aria-haspopup="listbox"][aria-label="${label}"]`);
  const options = (label) => [...document.querySelectorAll(`[role="listbox"][aria-label="${label}"] [role="option"]`)];
  window.__dd = {
    async count(label) {
      trigger(label).click();
      await sleep(150);
      const n = options(label).length;
      trigger(label).click();
      await sleep(150);
      return n;
    },
    async pick(label, text) {
      trigger(label).click();
      await sleep(150);
      const o = options(label).find((x) => x.textContent.trim() === text);
      if (!o) throw new Error(`${label}: "${text}" 없음`);
      o.click();
      await sleep(150);
    },
    value(label) {
      return trigger(label).textContent.replace("▾", "").trim(); // 트리거 끝에 ▾ 가 붙는다(실측)
    },
    leapBox() {
      return [...document.querySelectorAll("label")].some((l) => l.textContent.trim() === "윤달이야");
    },
    async calendar(name) {
      [...document.querySelectorAll("button")].find((b) => b.textContent.trim() === name).click();
      await sleep(150);
    },
  };
  return "ok";
})()
```
Expected: `"ok"`. 인앱 브라우저 screenshot 은 이 환경에서 타임아웃이 잦다 — 증거는 DOM 단정으로 남긴다.
